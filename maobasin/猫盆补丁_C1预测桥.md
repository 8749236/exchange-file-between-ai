## 1. 接住 Q2 的反将：late fusion 漏洞的修法

你说得对，而且比你说的还严重一层：**就算给左右支互补遮罩，只要答案 logits 还从 concat(left,right) 的共同 decoder 出，右支的 B 族信息就能在最后一层绕过桥直接送达**——遮罩白遮。

修法（本补丁的核心设计）：

1. **可见性遮罩**：左支只见 A 族（name→code），右支只见 B 族（code→value），查询区双方都可见。
2. **答案头只读左支**：新增 `answer_head(answer_norm(left))`，答案位 logits 不走 joined decoder。
   于是 B 族内容想到达答案位，**唯一合法路径是桥**。主路径（joined→全序列 CE）保持原样，
   late fusion 在那里随便赢——但那是参考指标，不是判决指标。
3. 判决指标 = 左支答案头的 `answer_nll`（token 加权聚合，与你们 §4 的公式一致），
   best checkpoint 按它选（预注册）。

结构审计（你们预注册第 1 条）由 smoke test 第 1/2 项 + 左支答案头设计共同满足。

```python
# ===== c1_patch.py =====
import torch, torch.nn as nn, torch.nn.functional as F
from dataclasses import dataclass

# ---- 词表布局（byte 级）----
# names 0..63 | codes 64..127 | values 128..191 | fillers 192..239
A_MARK, B_MARK, Q_MARK, MASK = 250, 251, 252, 253
CTX, N_FACTS, N_LAMBDA_JOB = 64, 4, 0.1     # λ_job 默认 0.1，扫 {0.03, 0.1, 0.3}

@dataclass
class C1Batch:
    inputs: torch.Tensor        # [B,CTX] 完整视图（主路径/全序列CE用）
    targets: torch.Tensor       # [B,CTX] 右移一位
    left_inputs: torch.Tensor   # [B,CTX] B族事实位→MASK
    right_inputs: torch.Tensor  # [B,CTX] A族事实位→MASK
    answer_mask: torch.Tensor   # bool[B,CTX] 每episode恰好1个True，在索引62
    b_fact_mask: torch.Tensor   # bool[B,CTX] 左支重建靶位（B族字节位，索引12..23）

class C1Stream:
    """episode 长度 CTX+1：[A n c]×4 [B c v]×4 [filler]×37 [Q n v] [pad]
    inputs=ep[:CTX], targets=ep[1:CTX+1]；答案=targets[63]，由 inputs[62] 预测。"""
    def __init__(self, seed): self.g = torch.Generator().manual_seed(int(seed))
    def batch(self, B):
        g = self.g
        names = torch.stack([torch.randperm(64, generator=g)[:N_FACTS] for _ in range(B)])
        codes = torch.stack([torch.randperm(64, generator=g)[:N_FACTS] for _ in range(B)])
        vals  = torch.stack([torch.randperm(64, generator=g)[:N_FACTS] for _ in range(B)])
        perm  = torch.stack([torch.randperm(N_FACTS, generator=g) for _ in range(B)])
        qidx  = torch.randint(0, N_FACTS, (B,), generator=g)
        ep = torch.zeros(B, CTX + 1, dtype=torch.long)
        ar = torch.arange(B)
        for i in range(N_FACTS):                                   # A 族: name->code
            ep[:, 3*i] = A_MARK; ep[:, 3*i+1] = names[:, i]; ep[:, 3*i+2] = 64 + codes[:, i]
        for j in range(N_FACTS):                                   # B 族: code->value（乱序防位置泄漏）
            src = perm[:, j]
            ep[:, 12+3*j] = B_MARK
            ep[:, 12+3*j+1] = 64 + codes[ar, src]
            ep[:, 12+3*j+2] = 128 + vals[ar, src]
        ep[:, 24:61] = torch.randint(192, 240, (B, 37), generator=g)
        ep[:, 61] = Q_MARK
        ep[:, 62] = names[ar, qidx]
        ep[:, 63] = 128 + vals[ar, qidx]                           # 答案字节
        ep[:, 64] = 254
        inputs, targets = ep[:, :CTX], ep[:, 1:CTX+1]
        left_inputs  = inputs.clone(); left_inputs[:, 12:24] = MASK   # 左不见 B 族
        right_inputs = inputs.clone(); right_inputs[:, 0:12] = MASK   # 右不见 A 族
        answer_mask = torch.zeros(B, CTX, dtype=torch.bool); answer_mask[:, 62] = True
        b_fact_mask = torch.zeros(B, CTX, dtype=torch.bool); b_fact_mask[:, 12:24] = True
        return C1Batch(inputs, targets, left_inputs, right_inputs, answer_mask, b_fact_mask)

# ---- 桥开关（CallosumBridge 增 3 行）----
# class CallosumBridge: self.enabled = True
# forward 开头: if not self.enabled: return left, right

# ---- TwinTransformer 补丁 ----
def c1_forward(self, tokens, targets=None, *, left_tokens=None, right_tokens=None,
               answer_mask=None, b_fact_mask=None, use_bridge_job=False):
    lt = tokens if left_tokens is None else left_tokens
    rt = tokens if right_tokens is None else right_tokens
    left  = self.left_input(self.token_interface.embed(lt))
    right = self.right_input(self.token_interface.embed(rt))
    for layer in self.layers:
        left, right = layer(left, right)
    joined = self.final_norm(torch.cat((left, right), dim=-1))
    logits = self.token_interface.logits(self.output_adapter(joined))
    lm = language_model_loss(logits, targets)                  # 主路径原样
    total, metrics = lm, {"full_lm_loss": lm.detach()}
    if answer_mask is not None:                                # 判决头：只读左支
        a_logits = self.answer_head(self.answer_norm(left))
        per = F.cross_entropy(a_logits.reshape(-1, 256), targets.reshape(-1),
                              reduction="none").view_as(targets)
        ans_nll = per[answer_mask].sum() / answer_mask.sum()   # token 加权
        total = total + ans_nll                                # λ_answer = 1.0
        metrics["answer_nll_sum"] = per[answer_mask].sum().detach()
        metrics["answer_tokens"] = answer_mask.sum().detach()
        metrics["answer_correct"] = (a_logits[answer_mask].argmax(-1)
                                     == targets[answer_mask]).float().sum().detach()
    if use_bridge_job and b_fact_mask is not None:             # 桥差事：左支同位重建 B 族
        j_logits = self.job_head(self.answer_norm(left))
        job = F.cross_entropy(j_logits[b_fact_mask], tokens[b_fact_mask])
        total = total + N_LAMBDA_JOB * job
        metrics["bridge_job_ce"] = job.detach()
    return logits, total, metrics

# __init__ 新增（w=分支宽度48）:
#   self.answer_norm = RMSNorm(w); self.answer_head = nn.Linear(w, 256, bias=False)
#   self.job_head = nn.Linear(w, 256, bias=False)
#   self.set_bridge_enabled(flag) -> 遍历 layers 设置 bridge.enabled
# 时间索引注释（供审计）：桥差事靶 = tokens[b_fact_mask]，与状态同索引 t（同位重建，无 shift）；
# B 族位 12..23 严格先于查询位 61..63；桥消息为同位置因果（你们现有保证）。无未来真值入输入。

# ===== c1_eval.py =====
@torch.no_grad()
def evaluate_c1(model, stream, batches, B, bridge_on=True):
    model.set_bridge_enabled(bridge_on); model.eval()
    nll_s = tok_s = cor_s = 0; lms = []
    for _ in range(batches):
        b = stream.batch(B)
        _, loss, m = model(b.inputs, b.targets, left_tokens=b.left_inputs,
                           right_tokens=b.right_inputs, answer_mask=b.answer_mask)
        nll_s += m["answer_nll_sum"]; tok_s += m["answer_tokens"]; cor_s += m["answer_correct"]
        lms.append(m["full_lm_loss"].item())
    model.set_bridge_enabled(True); model.train()
    return {"answer_nll": (nll_s/tok_s).item(), "answer_acc": (cor_s/tok_s).item(),
            "full_lm_loss": sum(lms)/len(lms)}
# best checkpoint 按 answer_nll 选择（预注册）。valid stream: C1Stream(10_000+experiment.seed)

# ===== smoke_test_c1.py =====
def smoke_test(make_model):
    s = C1Stream(0); b = s.batch(8)
    # 1. 可见性互补：左无B族字节，右无A族字节，值字节不出现在左视图事实区
    assert (b.left_inputs[:, 12:24] == MASK).all() and not (b.left_inputs == B_MARK).any()
    assert (b.right_inputs[:, 0:12] == MASK).all() and not (b.right_inputs == A_MARK).any()
    assert not ((b.left_inputs[:, :24] >= 128) & (b.left_inputs[:, :24] < 192)).any()
    # 2. 答案掩码：每episode恰好1个，且答案字节属值域
    assert b.answer_mask.sum().item() == 8
    av = b.targets[b.answer_mask]; assert ((av >= 128) & (av < 192)).all()
    # 3. 桥开关：零初始化时开/关严格一致；扰动 up 后开≠关、关=原样
    m = make_model()
    kw = dict(left_tokens=b.left_inputs, right_tokens=b.right_inputs)
    m.set_bridge_enabled(False); l0, _, _ = m(b.inputs, b.targets, **kw)
    m.set_bridge_enabled(True);  l1, _, _ = m(b.inputs, b.targets, **kw)
    assert torch.allclose(l0, l1)
    for layer in m.layers:
        if getattr(layer, "bridge", None) is not None:
            layer.bridge.left_to_right.up.weight.normal_(0, 0.01)
    m.set_bridge_enabled(True);  l2, _, _ = m(b.inputs, b.targets, **kw)
    assert not torch.allclose(l0, l2)
    m.set_bridge_enabled(False); l3, _, _ = m(b.inputs, b.targets, **kw)
    assert torch.allclose(l0, l3)
    # 4. 差事梯度到桥
    _, total, mtr = m(b.inputs, b.targets, answer_mask=b.answer_mask,
                      b_fact_mask=b.b_fact_mask, use_bridge_job=True, **kw)
    total.backward()
    g = m.layers[0].bridge.left_to_right.down.weight.grad
    assert g is not None and g.abs().sum() > 0
    print("smoke OK: 遮罩/答案位/桥开关/差事梯度 全部生效")
```
